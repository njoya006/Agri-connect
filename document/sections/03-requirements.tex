\section{Requirements Analysis}

\subsection{Functional Requirements}

\subsubsection{User Management}
\begin{itemize}[leftmargin=*]
	\item Users can register with email and password.
	\item Users can login and logout.
	\item Users can update their profiles.
	\item System supports farmer and admin roles.
\end{itemize}

\subsubsection{Farm Management}
\begin{itemize}[leftmargin=*]
	\item Farmers can add farms and fields.
	\item Farmers can log planting and harvesting.
	\item System shows activity history.
	\item Users can upload activity photos.
\end{itemize}

\subsubsection{Marketplace}
\begin{itemize}[leftmargin=*]
	\item Users can view price board.
	\item Users can create buy/sell listings.
	\item Users can search and filter listings.
	\item Users can message each other.
\end{itemize}
